```go
// hiddenColumns refuses a hidden column the display arranges or fits, and a
// hidden pivot cell column: each would show a column the result leaves out.
func (b *binder) hiddenColumns(p *plan) {
	for i, c := range p.spec.Columns {
		if !c.Hidden {
			continue
		}
		if pv := p.spec.Pivot; pv != nil && pv.Cell == c.ID {
			b.bad(fmt.Sprintf("columns.%d.hidden", i), "a pivot's cell column shows as the pivot's columns, so it is not hidden")
		}
		d := p.spec.Display
		if d == nil {
			continue
		}
		if _, ok := d.Columns[c.ID]; ok {
			b.bad("display.columns."+c.ID, "a hidden column is never shown, so it has no fit")
		}
		for li, l := range d.Layouts {
			for ci, cell := range l.Cells {
				if cell.Col == c.ID {
					b.bad(fmt.Sprintf("display.layouts.%d.cells.%d.col", li, ci), "a hidden column is never shown, so no layout places it")
				}
			}
		}
	}
}
```

**How it works (read before coding).** The SQL still computes a level-n row for a blank key at a skip level (its key `k(n-1)` is NULL, its cells total every row under it). That row is never emitted: `assemble` replaces it by its children, one depth less, and it is "open" exactly when its parent is. Its children keep their full group key, NULL included, so row ids, expansion keys and cursors keep their form. Because the hidden row's cells are the sum of what passes up, the parent's `"recompute"` cells already count the passed rows, and its `"rows"` cells add the hidden row's sum (its shown children). The blank key at a skip level is shown (in `s<L>`/`rt<L>`) only when it has a shown child, so a pass-through with nothing to pass is no child of its parent (G5 relies on this).

```go
	col string
	e   expr
}

// headings compiles every column's heading; a heading on an unset param is
// left out, so the column shows its label.
func (c *compiler) headings(s *stmt) ([]headingPlan, error) {
	var out []headingPlan
	for _, cp := range c.p.cols {
		n := cp.col.Heading
		if n == nil {
			continue
		}
		at := cp.at + ".heading"
		e, err := compileFormula(&headingCtx{c: c, s: s}, n, at)
		if err != nil {
			return nil, err
		}
		switch e.typ {
		case Text, Date, DateTime, Number, "":
		default:
			return nil, invalid(at, "a heading is a text, a day or a number")
		}
		if !e.unset {
			out = append(out, headingPlan{col: cp.col.ID, e: e})
		}
	}
	return out, nil
}

// readHeadings computes the run's headings in one statement.
func (rn *runner) readHeadings() error {
	s := newStmt(rn.app, rn.c.info)
	plans, err := rn.c.headings(s)
	if err != nil || len(plans) == 0 {
		return err
	}
	cells := make([]string, len(plans))
	for i, h := range plans {
		cells[i] = fmt.Sprintf("%s AS h%d", h.e.sql, i)
	}
	rows, err := rn.query(s, nil, "SELECT "+strings.Join(cells, ", "))
	if err != nil || len(rows) == 0 {
		return err
	}
	rn.headings = map[string]*ColumnHeading{}
	for i, h := range plans {
		v := rows[0][fmt.Sprintf("h%d", i)]
		if v == nil {
			continue
		}
		typ := h.e.typ
		if typ == "" {
			typ = Text
		}
		if typ == Date {
			// A day reads as YYYY-MM-DD, as a date param's value does.
			if text, ok := v.(string); ok && len(text) >= 10 {
				v = text[:10]
			}
		}
		rn.headings[h.col] = &ColumnHeading{Value: v, Type: typ}
	}
	return nil
}
```

## Rulings

1. **Skip is refused on the first level too** (spec names only the last level and a bucket). A blank first key has no group above to sit under (the Total is not a group, and a layout's sections name first-level keys); allowing it would need an "always open" blank top-level key that the expansion cannot name. The Balance Sheet skips only levels 2 and 3.
2. **A skipped key keeps its place in the group key.** Rows passed up keep their full key with `null` at the skipped level, so row ids, `options.expand` keys and page cursors keep one form; the blank key at a skip level is open exactly when its parent is (SQL `expansion.where`, Go `expansion.has`). The cursor fingerprint includes `blank`, so a cursor from before a change to it is refused.
3. **A skip level's having does not apply to its blank key.** The blank key is no group; it passes when something under it shows. This is also what makes G5's "no direct child rows from a skipped level" exact.
4. **G5 hides exactly as a having hides.** A `"rows"` total above a hidden group adds nothing it did not show; a `"recompute"` total recomputes from every base row the filter keeps, as it already does under any having (docs say so).
