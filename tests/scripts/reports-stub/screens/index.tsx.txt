// E2E fixture screen: adds teams, sales and returns through the UI, as the reports specs
// must set their data up.

const teamSchema = z.object({
    name: z.string().trim().min(1).max(200),
    region: z.enum(['north', 'south', 'east', '']),
})
type TeamValues = z.infer<typeof teamSchema>
const saleSchema = z.object({
    team: z.string().min(1, 'Choose a team'),
    customer: z.string().trim().min(1).max(200),
    amount: z.number().int(),
    cost: z.number().int(),
    quantity: z.number().int(),
    day: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'Enter a date as YYYY-MM-DD'),
    status: z.enum(['open', 'paid']),
})
type SaleValues = z.infer<typeof saleSchema>
const manySalesSchema = z.object({
    team: z.string().min(1, 'Choose a team'),
    prefix: z.string().trim().min(1).max(190),
    count: z.number().int().min(1).max(100),
    firstDay: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'Enter a date as YYYY-MM-DD'),
    firstAmount: z.number().int(),
    cost: z.number().int(),
    quantity: z.number().int(),
})
type ManySalesValues = z.infer<typeof manySalesSchema>
const returnSchema = z.object({
    team: z.string().min(1, 'Choose a team'),
    customer: z.string().trim().min(1).max(200),
    amount: z.number().int(),
    day: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'Enter a date as YYYY-MM-DD'),
})
type ReturnValues = z.infer<typeof returnSchema>
// A whole data set in one save. A sale and a return name their team; the team is
// one of the set's own or one saved before.
const dataSetSchema = z.object({
    label: z.string().min(1),
    teams: z.array(teamSchema).default([]),
    sales: z.array(saleSchema).default([]),
    returns: z.array(returnSchema).default([]),
})
function parseDataSet(json: string) {
    try {
        return dataSetSchema.safeParse(JSON.parse(json))
    } catch {
        return undefined
    }
}
const dataSetFormSchema = z.object({
    json: z
        .string()
        .min(1)
        .refine(json => parseDataSet(json)?.success === true, 'Enter a data set as JSON'),
})
type DataSetFormValues = z.infer<typeof dataSetFormSchema>

const REGION_OPTIONS = [
    { value: 'north', label: 'North' },
    { value: 'south', label: 'South' },
    { value: 'east', label: 'East' },
    // A team with no region is in the blank group of a list grouped by region.
    { value: '', label: 'No region' },
]

const STATUS_OPTIONS = [
    { value: 'open', label: 'Open' },
    { value: 'paid', label: 'Paid' },
]

function SavedNote({ text, testID }: { text: string; testID: string }) {
    if (!text) return null
    return (
        <Text testID={testID} className="text-sm text-muted-foreground">
            {text}
        </Text>
    )
}

function TeamForm() {
    const [teams] = useStore('salesstub_teams')
    const [saved, setSaved] = useState('')
    const form = useForm<TeamValues>({
        resolver: zodResolver(teamSchema),
        mode: 'onChange',
        defaultValues: { name: '', region: 'north' },
    })
    const create = useMutation({
        mutationFn: function* (values: TeamValues) {
            yield teams.insert({ id: newRecordId(), name: values.name, region: values.region })
        },
        onSuccess: (_data, values) => {
            setSaved(`Saved ${values.name}`)
            form.reset({ name: '', region: values.region })
        },
    })
    const onCreate = () => {
        void form.handleSubmit(values => create.mutate(values))()
    }
    return (
        <View className="gap-2">
            <Text className="text-base font-semibold text-foreground">New team</Text>
            <TextInput
                name="name"
                control={form.control}
                label="Team name"
                testID="salesstub-team-name"
            />
            <ButtonGroup
                name="region"
                control={form.control}
                label="Region"
                horizontal
                options={REGION_OPTIONS}
            />
            <Button onPress={onCreate} isDisabled={create.isPending} testID="salesstub-team-create">
                <ButtonText>Add team</ButtonText>
            </Button>
            <SavedNote text={saved} testID="salesstub-team-saved" />
        </View>
    )
}

// The day `offset` days after `first` (YYYY-MM-DD), in UTC.
function dayAfter(first: string, offset: number): string {
    const [y, m, d] = first.split('-').map(Number)
    return new Date(Date.UTC(y, m - 1, d + offset)).toISOString().slice(0, 10)
}

function useTeamOptions() {
    const [teams] = useStore('salesstub_teams')
    const { data: teamRows } = useLiveQuery({
        queryKey: ['salesstub.teams'],
        query: q => q.from({ team: teams }).orderBy(({ team }) => team.name),
    })
    return (teamRows ?? []).map(team => ({ value: team.id, label: team.name }))
}

// Many open sales in one save, for a spec that needs a long list: customers
// "<prefix> 00", "<prefix> 01", …, one day apart, each amount one more.
function ManySalesForm() {
    const { user } = useAuth()
    const [sales] = useStore('salesstub_sales')
    const [saved, setSaved] = useState('')
    const teamOptions = useTeamOptions()
    const form = useForm<ManySalesValues>({
        resolver: zodResolver(manySalesSchema),
        mode: 'onChange',
        defaultValues: {
            team: '',
            prefix: '',
            count: 1,
            firstDay: '',
            firstAmount: 0,
            cost: 0,
            quantity: 0,
        },
    })
    const create = useMutation({
        mutationFn: function* (values: ManySalesValues) {
            const rows = Array.from({ length: values.count }, (_, i) => {
                const day = dayAfter(values.firstDay, i)
                return sales.insert({
                    id: newRecordId(),
                    owner: user.id,
                    team: values.team,
                    customer: `${values.prefix} ${String(i).padStart(2, '0')}`,
                    amount: values.firstAmount + i,
                    cost: values.cost,
                    quantity: values.quantity,
                    currency: 'USD',
                    day: `${day} 00:00:00.000Z`,
                    booked_on: day,
                    status: 'open',
                })
            })
            yield rows
        },
        onSuccess: (_data, values) => setSaved(`Saved ${values.count} sales`),
    })
    const onCreate = () => {
        void form.handleSubmit(values => create.mutate(values))()
    }
    return (
        <View className="gap-2">
            <Text className="text-base font-semibold text-foreground">Many sales</Text>
            <SelectInput
                name="team"
                control={form.control}
                label="Team"
                options={teamOptions}
                testID="salesstub-sales-team"
            />
            <TextInput
                name="prefix"
                control={form.control}
                label="Customer prefix"
                testID="salesstub-sales-prefix"
            />
            <NumberInput
                name="count"
                control={form.control}
                label="How many"
                testID="salesstub-sales-count"
            />
            <TextInput
                name="firstDay"
                control={form.control}
                label="First day"
                placeholder="YYYY-MM-DD"
                testID="salesstub-sales-first-day"
            />
            <NumberInput
                name="firstAmount"
                control={form.control}
                label="First amount (cents)"
                testID="salesstub-sales-first-amount"
            />
            <NumberInput
                name="cost"
                control={form.control}
                label="Cost (cents)"
                testID="salesstub-sales-cost"
            />
            <NumberInput
                name="quantity"
                control={form.control}
                label="Quantity"
                testID="salesstub-sales-quantity"
            />
            <Button
                onPress={onCreate}
                isDisabled={create.isPending}
                testID="salesstub-sales-create"
            >
                <ButtonText>Add sales</ButtonText>
            </Button>
            <SavedNote text={saved} testID="salesstub-sales-saved" />
        </View>
    )
}

// Teams, sales and returns in one save, for a spec's whole seed: each screen
// change and each form round is most of a second of a spec's time. The rows are
// saved one after the other, in the order given, so their record order is the
// set's order.
function DataSetForm() {
    const { user } = useAuth()
    const [teams, sales, returns] = useStore(
        'salesstub_teams',
        'salesstub_sales',
        'salesstub_returns'
    )
    const [saved, setSaved] = useState('')
    const teamOptions = useTeamOptions()
    const form = useForm<DataSetFormValues>({
        resolver: zodResolver(dataSetFormSchema),
        mode: 'onChange',
        defaultValues: { json: '' },
    })
    const create = useMutation({
        mutationFn: function* (values: DataSetFormValues) {
            const data = dataSetSchema.parse(JSON.parse(values.json))
            const teamIds = new Map(teamOptions.map(option => [option.label, option.value]))
            const teamId = (name: string) => {
                const id = teamIds.get(name)
                if (!id) throw new Error(`no team named ${name}`)
                return id
            }
            for (const team of data.teams) {
                const id = newRecordId()
                teamIds.set(team.name, id)
                yield teams.insert({ id, name: team.name, region: team.region })
            }
            for (const sale of data.sales) {
                yield sales.insert({
                    id: newRecordId(),
                    owner: user.id,
                    team: teamId(sale.team),
                    customer: sale.customer,
                    amount: sale.amount,
                    cost: sale.cost,
                    quantity: sale.quantity,
                    currency: 'USD',
                    day: `${sale.day} 00:00:00.000Z`,
                    booked_on: sale.day,
                    status: sale.status,
                })
            }
            for (const ret of data.returns) {
                yield returns.insert({
                    id: newRecordId(),
                    owner: user.id,
                    team: teamId(ret.team),
                    customer: ret.customer,
                    amount: ret.amount,
                    currency: 'USD',
                    day: `${ret.day} 00:00:00.000Z`,
                })
            }
            return data.label
        },
        onSuccess: label => {
            setSaved(`Saved ${label}`)
            form.reset({ json: '' })
        },
        onError: error => setSaved(`Not saved: ${error.message}`),
    })
    const onCreate = () => {
        void form.handleSubmit(values => create.mutate(values))()
    }
    return (
        <View className="gap-2">
            <Text className="text-base font-semibold text-foreground">Data set</Text>
            <TextInput
                name="json"
                control={form.control}
                label="Teams, sales and returns (JSON)"
                testID="salesstub-data-json"
            />
            <Button onPress={onCreate} isDisabled={create.isPending} testID="salesstub-data-create">
                <ButtonText>Add data set</ButtonText>
            </Button>
            <SavedNote text={saved} testID="salesstub-data-saved" />
        </View>
    )
}

const DRAWER_MENU_CHOICES = ['North', 'South', 'East']

// A core Menu (default `auto` presentation) inside a core overlay Drawer, for
// tests/e2e/sheet-in-drawer.spec.ts: on a phone the Menu is a Sheet that must
// draw above the Drawer it opens from; on a wide window it is a popover.
function DrawerMenuDemo() {
    const [isOpen, setIsOpen] = useState(false)
    const [chosen, setChosen] = useState('None')
    const items = DRAWER_MENU_CHOICES.map(label => (
        <Menu.Item
            key={label}
            label={label}
            isSelected={label === chosen}
            onSelect={() => setChosen(label)}
            testID={`salesstub-drawer-menu-${label.toLowerCase()}`}
        />
    ))
    return (
        <View className="gap-2">
            <Button onPress={() => setIsOpen(true)} testID="salesstub-drawer-open">
                <ButtonText>Open drawer</ButtonText>
            </Button>
            <Drawer isOpen={isOpen} onClose={() => setIsOpen(false)} anchor="right">
                <DrawerBackdrop />
                <DrawerContent testID="salesstub-drawer">
                    <DrawerHeader>
                        <Text className="text-lg font-semibold text-foreground">Stub drawer</Text>
                    </DrawerHeader>
                    <DrawerBody>
                        <Text testID="salesstub-drawer-chosen" className="text-foreground">
                            {chosen}
                        </Text>
                        <Menu
                            title="Region"
                            testID="salesstub-drawer-menu"
                            trigger={
                                <Button testID="salesstub-drawer-menu-trigger">
                                    <ButtonText>Choose region</ButtonText>
                                </Button>
                            }
                        >
                            {items}
                        </Menu>
                    </DrawerBody>
                </DrawerContent>
            </Drawer>
        </View>
    )
}
