// Package stub is an E2E fixture: two report sources over its own collections,
// so the app shell's reports e2e has a to-one link, a to-many link, money, a
// number, a date, a day kept as text and selects with no real feature package
// installed, a heading, and two shipped reports (one published, one not). See
// tinycld/tests/scripts/scaffold-reports-stub.ts for why.
package stub

import (
	"encoding/json"

	"github.com/pocketbase/pocketbase/core"

	"tinycld.org/core/reports"
)

// Register is the entry point the generated package_extensions.go calls.
func Register(_ core.App) error {
	reports.RegisterSource(reports.Source{
		Key: "salesstub.sales", Collection: "salesstub_sales", Label: "Stub sales",
		Fields: []reports.Field{
			{Key: "customer", Label: "Customer", Type: reports.Text, Aggregates: reports.CountOnly},
			{Key: "amount", Label: "Amount", Type: reports.Money, Aggregates: reports.SumAll, Currency: "currency"},
			{Key: "cost", Label: "Cost", Type: reports.Money, Aggregates: reports.SumAll, Currency: "currency"},
			{Key: "quantity", Label: "Quantity", Type: reports.Number, Aggregates: reports.SumAll},
			{Key: "currency", Label: "Currency", Type: reports.Text},
			{Key: "day", Label: "Day", Type: reports.Date, Aggregates: reports.MinMax},
			// A day kept as text, read as a date (E1).
			{Key: "booked", Column: "booked_on", Label: "Booked", Type: reports.Date, TextDay: true, Aggregates: reports.MinMax},
			{Key: "status", Label: "Status", Type: reports.Select, Options: []reports.Option{
				{Value: "open", Label: "Open"}, {Value: "paid", Label: "Paid"},
			}},
		},
		Links: []reports.Link{
			{Key: "team", Label: "Team", Target: "salesstub.teams", Cardinality: reports.ToOne, Field: "team"},
		},
		Heading: stubHeading,
	})
	reports.RegisterSource(reports.Source{
		Key: "salesstub.teams", Collection: "salesstub_teams", Label: "Stub teams",
		Fields: []reports.Field{
			{Key: "name", Label: "Name", Type: reports.Text, Aggregates: reports.MinMax},
			// Declared North, South, East: not A–Z, so a picker in the field's
			// own order differs from one by name (optionsFrom.groupOrder).
			{Key: "region", Label: "Region", Type: reports.Select, Options: []reports.Option{
				{Value: "north", Label: "North"}, {Value: "south", Label: "South"}, {Value: "east", Label: "East"},
			}},
		},
		Links: []reports.Link{
			{Key: "sales", Label: "Sales", Target: "salesstub.sales", Cardinality: reports.ToMany, BackField: "team"},
		},
		Heading: stubHeading,
	})
	// A shipped report whose formula uses `if`, which the builder cannot edit, so the
	// e2e covers a published template, its read-only formula and a copy saved
	// unchanged (tests/e2e/reports-shipped.spec.ts asserts this formula).
	reports.RegisterShipped(reports.Shipped{
		Key: "salesstub.larger_side", Title: "Stub larger side", Revision: 1,
		Description: "Each sale's amount or cost, whichever is larger.",
		Spec: json.RawMessage(`{
			"version": 1,
			"source": "salesstub.sales",
			"columns": [
				{"id": "customer", "path": [{"field": "customer"}]},
				{"id": "amount", "path": [{"field": "amount"}]},
				{"id": "cost", "path": [{"field": "cost"}]},
				{"id": "larger", "label": "Larger", "formula": {"op": "if", "args": [
					{"op": "gt", "args": [{"col": "amount"}, {"col": "cost"}]},
					{"col": "amount"},
					{"col": "cost"}
				]}}
			]
		}`),
	})
	// An unpublished shipped report (E10) that uses the v2 parts: a record
	// group with a label, a total of the rows shown, a sort formula, a detail
	// label, a text-day filter and a choice from a source grouped by region,
	// whose group pick has its own text ("All North teams").
	// The e2e owner shares it to show who sees it (reports-source-params.spec.ts).
	reports.RegisterShipped(reports.Shipped{
		Key: "salesstub.team_totals", Title: "Stub team totals", Revision: 2, Unpublished: true,
		Description: "Each team's sales booked on or before a day.",
		Spec: json.RawMessage(`{
			"version": 2,
			"source": "salesstub.sales",
			"params": [
				{"id": "asOf", "type": "asOf", "label": "As of", "default": {"preset": "today"}},
				{"id": "teams", "type": "choice", "label": "Teams", "multiple": true,
				 "optionsFrom": {"source": "salesstub.teams", "value": [{"field": "id"}], "label": [{"field": "name"}], "group": [{"field": "region"}], "groupLabel": "All {group} teams"}}
			],
			"columns": [
				{"id": "team", "label": "Team", "path": [{"link": "team"}]},
				{"id": "region", "label": "Region", "formula": {"record": "byTeam", "path": [{"field": "region"}]}},
				{"id": "amount", "label": "Amount", "path": [{"field": "amount"}], "aggregate": "sum", "total": "rows"}
			],
			"filter": {"op": "and", "args": [
				{"op": "lte", "args": [{"path": [{"field": "booked"}]}, {"param": "asOf"}]},
				{"op": "in", "args": [{"path": [{"link": "team"}]}, {"param": "teams"}]}
			]},
			"groups": [{"id": "byTeam", "col": "team", "label": {"record": "byTeam", "path": [{"field": "name"}]}}],
			"sort": [{"formula": {"record": "byTeam", "path": [{"field": "name"}]}}],
			"detail": {"label": {"op": "concat", "args": [{"path": [{"field": "customer"}]}, {"value": " · "}, {"path": [{"field": "booked"}]}]}}
		}`),
	})
	return nil
}

// stubHeading is the business name the stub's reports print above their title (E9).
func stubHeading(core.App, *core.RequestInfo) string { return "Stub Sales Co" }
